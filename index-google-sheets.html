Código do Google Apps Script

1. Crie uma Planilha no Google (sheets.new) e vá em Extensões > Apps Script.

2. Substitua o código pelo script abaixo e clique em Salvar.

3. Clique em Implantar > Nova Implantação > Selecione App da Web:

↳ Executar como: Eu | Quem tem acesso: Qualquer pessoa

4. Copie a URL gerada e cole no campo "Conectar Planilha".

function doGet(e) {
  var action = (e && e.parameter && e.parameter.action) || 'getData';
  var callback = e && e.parameter && e.parameter.callback;
  var responseData = {};
  
  if (action === 'getResponses') {
    responseData = { result: "success", responses: getResponsesFromSheet() };
  } else {
    responseData = {
      result: "success",
      settings: getSettings(),
      questions: getQuestionsFromSheet()
    };
  }
  
  var jsonString = JSON.stringify(responseData);
  if (callback) {
    return ContentService.createTextOutput(callback + '(' + jsonString + ');').setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(jsonString).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var action = data.action || 'submitResponse';
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (action === 'saveQuestions') {
      if (data.questions) saveQuestionsToSheet(data.questions);
      if (data.settings) saveSettings(data.settings);
      return ContentService.createTextOutput(JSON.stringify({result:"success"})).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === 'saveSettings') {
      saveSettings(data.settings);
      return ContentService.createTextOutput(JSON.stringify({result:"success"})).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === 'clearResponses') {
      clearResponsesSheet();
      return ContentService.createTextOutput(JSON.stringify({result:"success"})).setMimeType(ContentService.MimeType.JSON);
    }
    
    var sheet = getOrCreateSheet(ss, "Respostas");
    sheet.appendRow([
      data.date || new Date().toLocaleString('pt-BR'),
      data.nome || '',
      data.sobrenome || '',
      data.acertos || '0',
      JSON.stringify(data.respostas || {})
    ]);
    return ContentService.createTextOutput(JSON.stringify({result:"success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result:"error", error: err.message})).setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    if (name === "Respostas") {
      sheet.appendRow(["Data", "Nome", "Sobrenome", "Acertos", "JSON Respostas"]);
      sheet.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#dcfce7");
    } else if (name === "Perguntas") {
      sheet.appendRow(["ID", "Enunciado", "Opcao_A", "Opcao_B", "Opcao_C", "Opcao_D", "Gabarito", "Liberada"]);
      sheet.getRange(1, 1, 1, 8).setFontWeight("bold").setBackground("#e0e7ff");
    }
  }
  return sheet;
}

function getSettings() {
  var props = PropertiesService.getScriptProperties();
  return {
    lessonTitle: props.getProperty("LESSON_TITLE") || "Simulado Geral",
    backgroundDriveUrl: props.getProperty("BG_URL") || ""
  };
}

function saveSettings(settings) {
  if (!settings) return;
  var props = PropertiesService.getScriptProperties();
  if (settings.lessonTitle) props.setProperty("LESSON_TITLE", settings.lessonTitle);
  if (settings.backgroundDriveUrl !== undefined) props.setProperty("BG_URL", settings.backgroundDriveUrl);
}

function getQuestionsFromSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateSheet(ss, "Perguntas");
  var values = sheet.getDataRange().getValues();
  var questions = [];
  
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (!row[0] && !row[1]) continue;
    var visVal = String(row[7]).toLowerCase().trim();
    questions.push({
      id: Number(row[0]) || (Date.now() + i),
      title: String(row[1] || ''),
      options: { A: String(row[2] || ''), B: String(row[3] || ''), C: String(row[4] || ''), D: String(row[5] || '') },
      correct: String(row[6] || 'A').trim().toUpperCase(),
      visible: (visVal === 'true' || visVal === 'sim' || row[7] === true || visVal === '1')
    });
  }
  if (questions.length === 0) {
    var defaultQ = { id: 1, title: "Qual é a capital do Brasil?", options: { A: "Rio de Janeiro", B: "São Paulo", C: "Brasília", D: "Salvador" }, correct: "C", visible: true };
    saveQuestionsToSheet([defaultQ]);
    return [defaultQ];
  }
  return questions;
}

function saveQuestionsToSheet(questions) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateSheet(ss, "Perguntas");
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, 8).clearContent();
  if (!questions || questions.length === 0) return;
  
  var rows = questions.map(function(q) {
    return [q.id, q.title, q.options ? q.options.A : '', q.options ? q.options.B : '', q.options ? q.options.C : '', q.options ? q.options.D : '', q.correct || 'A', q.visible ? 'TRUE' : 'FALSE'];
  });
  sheet.getRange(2, 1, rows.length, 8).setValues(rows);
}

function getResponsesFromSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateSheet(ss, "Respostas");
  var values = sheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (!row[0] && !row[1]) continue;
    var respObj = {};
    try { respObj = JSON.parse(row[4]); } catch(e) {}
    list.push({ date: String(row[0]), nome: String(row[1]), sobrenome: String(row[2]), acertos: String(row[3]), respostas: respObj });
  }
  return list;
}

function clearResponsesSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateSheet(ss, "Respostas");
  var lastRow = sheet.getLastRow();
  if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, 5).clearContent();
}
